   └─ Outros (com validações de acesso)
   ↓
RESPOSTA FILTRADA POR PERMISSÕES

┌─────────────────────────────────────────────────────────────────────────────┐
│                       🔑 TIPOS DE EMPRESA E NÍVEIS                          │
└─────────────────────────────────────────────────────────────────────────────┘

MICROEMPRESA (ME) - emp_tipo = 0
├─ Abas: Dashboard, Documentos, Impostos, Faturamento, Caixa, Prazos, Perfil
├─ Visualizador (0):     Visualiza tudo
├─ Gerente (1):          Edita dados, sem upload
└─ ADM (2):              Acesso total + uploads ⭐

MICROEMPREENDEDOR (MEI) - emp_tipo = 1
├─ Abas: Dashboard, Imposto (DAS), Notas Emitidas, Controle Mensal
├─ Visualizador (0):     Sem acesso a DAS ❌
├─ Gerente (1):          Acesso a DAS e Notas, edita
└─ ADM (2):              Acesso total ⭐

┌─────────────────────────────────────────────────────────────────────────────┐
│                        📋 FLUXO DE AUTENTICAÇÃO                             │
└─────────────────────────────────────────────────────────────────────────────┘

1️⃣  LOGIN
    GET /usuarios/login?email=xxx&senha=xxx
    ↓ Retorna:
    {
      "usuario": { "id": 1, "nome": "João" },
      "empresas": [...],
      "empresa_padrao": {
        "emp_id": 5,
        "tipo": "ME",
        "nivel_acesso": 1,
        "nivel_descricao": "Gerente"
      }
    }

2️⃣  USAR TOKEN NAS REQUISIÇÕES
    GET /api/dashboard/resumo
    Header: Authorization: {"usuarioId": 1, "empresaId": 5}

3️⃣  MIDDLEWARE VALIDA
    ✓ Usuário existe e está ativo
    ✓ Empresa existe e está ativa
    ✓ Vínculo entre usuário e empresa existe
    ✓ Carrega tipo de empresa e nível de acesso

4️⃣  RETORNA DADOS FILTRADOS
    {
      "empresa": {"id": 5, "tipo": "ME"},
      "documentos": 15,
      "prazos_pendentes": 3,
      "financeiro": {...}
    }

┌─────────────────────────────────────────────────────────────────────────────┐
│                      💡 EXEMPLOS DE USO (JAVASCRIPT)                        │
└─────────────────────────────────────────────────────────────────────────────┘

// 1. LOGIN
const login = await fetch('/usuarios/login?email=user@email.com&senha=123');
const data = await login.json();
const { usuario, empresa_padrao } = data.dados;

// 2. GUARDAR AUTH
const auth = {
  usuarioId: usuario.id,
  empresaId: empresa_padrao.emp_id
};

// 3. USAR NAS REQUISIÇÕES
const dashboard = await fetch('/api/dashboard/resumo', {
  headers: {
    'Authorization': JSON.stringify(auth)
  }
});

// 4. FAZER UPLOAD (se ADM)
const formData = new FormData();
formData.append('tpd_id', 1);
formData.append('arquivo', file);

await fetch('/api/documentos/autenticados', {
  method: 'POST',
  headers: { 'Authorization': JSON.stringify(auth) },
  body: formData
});
